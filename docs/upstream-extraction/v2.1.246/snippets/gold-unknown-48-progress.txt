function o2o(e, t) {
  // TaskCreate (schema n2o: subject, activeForm?)
  //   pendingCreates.set(t.id, {content:subject, activeForm:activeForm??subject, status:"pending"})
  // TaskUpdate Cp (schema r2o: taskId, status?, subject?, activeForm?)
  //   status==="deleted" → tasks.delete
  //   else merge {content, activeForm, status} (defaults keep previous / pending)
  // else return false
}

function s2o(e, t) {
  // tool_result for a pending TaskCreate
  let n = e.pendingCreates.get(t.tool_use_id)
  if (!n) return false
  if (t.is_error) { e.pendingCreates.delete(t.tool_use_id); return true }
  // t2o = /^Task #(\S+) created successfully/
  let o = content.match(t2o)?.[1]
  if (!o) return false
  e.pendingCreates.delete(t.tool_use_id)
  if (!e.tasks.has(o)) e.tasks.set(o, n)
  return true
}

function i2o(e) {
  return [...e.tasks.values(), ...e.pendingCreates.values()]
}

function pGn() {
  let e = { pendingCreates: new Map, tasks: new Map }, t, n
  return {
    observe(r) {
      if (r.type === "assistant" && !r.parent_tool_use_id) {
        for (let s of r.message.content)
          if (s.type === "tool_use" && o2o(e, s)) n = void 0
        let o = r.message.content.find(s => s.type === "tool_use" && s.name === pf)
        if (o) t = { ...r, message: { ...r.message, content: [o] } }, n = void 0
      } else if (r.type === "user") {
        let o = r.message.content
        if (!Array.isArray(o)) return
        for (let s of o)
          if (s.type === "tool_result" && s2o(e, s)) n = void 0
      }
    },
    todos() {
      if (n === void 0) {
        let r = i2o(e)
        n = r.length > 0 ? r : t ? e2o([t]) : []
      }
      return n
    },
  }
}
