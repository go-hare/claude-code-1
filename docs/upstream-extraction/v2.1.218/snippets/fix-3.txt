### L6s — recursive unescape with Windows path skip (THE FIX)

```js
function L6s(e, t) {
  if (typeof e === "string") {
    if (!e.includes("\\u")) return e;
    if (!qYd.test(e)) return e;
    // densable 2.1.218: leave \u-prefixed path segments literal (e.g. C:\Users\u4e2d)
    if (Cky.test(e)) {
      t.windowsPathSkips++;
      return e;
    }
    let r = e.replace(vky, (n, o, i, s, a) => {
      // odd number of preceding backslashes => already escaped; keep match
      let l = a;
      while (l > 0 && e[l - 1] === "\\") l--;
      if ((a - l) & 1) return n;
      // surrogate pair: \uD800-\uDBFF \uDC00-\uDFFF
      if (o !== void 0)
        return String.fromCharCode(parseInt(o, 16), parseInt(i, 16));
      let c = parseInt(s, 16);
      // lone surrogate: keep literal (do not emit invalid UTF-16)
      if (c >= 55296 && c <= 57343) return n;
      return String.fromCharCode(c);
    });
    if (r !== e) t.repairedStrings++;
    return r;
  }
  if (Array.isArray(e)) return e.map((r) => L6s(r, t));
  if (e !== null && typeof e === "object") {
    let r = {};
    for (let [n, o] of Object.entries(e)) r[n] = L6s(o, t);
    return r;
  }
  return e;
}
```

## Bug semantics

After `JSON.parse` of tool_use.input, a Windows path like
`C:\Users\u4e2d\file` is a JS string whose content still contains the two
characters `\` + `u` + four hex digits. A naive double-escaped-unicode repair
(`String.fromCharCode(parseInt(...))`) turns `\u4e2d` into CJK `中`, making the
path inaccessible. densable `Cky` short-circuits the whole string when it looks
like a Windows drive/UNC path so those `\u…` path segments stay literal.

`C:\Users\unicorn` in the changelog is the illustrative `\u`-prefixed segment
shape (not all four following chars need be hex for the product description).
