### hFt (extractDangerousSettings)
```js
function hFt(e) {
  if (!e) return { shellSettings: {}, envVars: {}, hasHooks: false, hasClaudeMd: false }
  let t = {}
  for (let i of S3l) {
    let s = e[i], a
    if (typeof s === "string") a = s
    else if (s !== null && typeof s === "object" && "command" in s && typeof s.command === "string") a = s.command
    if (a !== undefined && a.length > 0) t[i] = a
  }
  let r = {}
  if (e.env && typeof e.env === "object")
    for (let [i, s] of Object.entries(e.env)) {
      if (s === undefined) continue
      let a = String(s)
      // benign feature/cost toggles excluded here via B7t → no approval
      if (a.length > 0 && !B7t(i, a)) r[i] = a
    }
  let n = e.hooks !== undefined && e.hooks !== null && typeof e.hooks === "object" && Object.keys(e.hooks).length > 0
  let o = typeof e.claudeMd === "string" && e.claudeMd.length > 0
  return {
    shellSettings: t,
    envVars: r,
    hasHooks: n,
    hooks: n ? e.hooks : undefined,
    hasClaudeMd: o,
    claudeMd: o ? e.claudeMd : undefined,
  }
}
```
