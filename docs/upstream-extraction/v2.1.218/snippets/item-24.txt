## densable bodies (2.1.218 SEA, verbatim)

### q0 — string-array sanitize (the 218 delta fix core)

```js
function q0(e){
  if(!Array.isArray(e))return[];
  return e.every((t)=>typeof t==="string")?e:e.filter((t)=>typeof t==="string")
}
```

Used on every delta string[] field before `.length` / `.join` / `for…of` so a missing/null/non-array/mixed-type field cannot throw (resume every-turn fail / crash).

### qQu / gen — structural attachment drop (already in 2.1.217; unchanged body in 218)

```js
function qQu(e){
  if(typeof e!=="object"||e===null||!("type"in e)||typeof e.type!=="string")return!1;
  switch(e.type){
    case"new_file":return"filename"in e&&typeof e.filename==="string";
    case"new_directory":return"path"in e&&typeof e.path==="string";
    case"invoked_skills":return"skills"in e&&Array.isArray(e.skills)&&e.skills.every((t)=>typeof t==="object"&&t!==null);
    case"hook_success":return"content"in e&&typeof e.content==="string";
    case"skill_listing":return!("names"in e)||e.names===void 0||Array.isArray(e.names);
    case"hook_additional_context":return"content"in e&&Array.isArray(e.content)&&e.content.every((t)=>typeof t==="string");
    default:return!0
  }
}
function gen(e){
  let t=0,r=e.filter((n)=>{
    if(n.type==="attachment"&&!qQu(n.attachment))return t+=1,!1;
    return!0
  });
  if(t===0)return e;
  return C(`resume: dropped ${t} attachment ${St(t,"entry","entries")} with a missing or malformed payload \u2014 the session transcript appears partially corrupt`,{level:"error"}),r
}
```

Note: qQu `default: return true` — delta types pass the structural gate; malformed *fields inside* deltas are handled by q0, not by gen.
