// DS / GA / Z2o — slice from last compact_boundary
function GA(e){return e?.type==="system"&&e.subtype==="compact_boundary"}
function Z2o(e){for(let t=e.length-1;t>=0;t--){let r=e[t];if(r&&GA(r))return t}return-1}

  return{...e,message:{...e.message,usage:{...e.message.usage,
