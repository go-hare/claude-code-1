function isDeferredTool(tool: {
  name: string
  alwaysLoad?: boolean
  isMcp?: boolean
  shouldDefer?: boolean
}): boolean {
  // 1) alwaysLoad forces non-deferred
  if (tool.alwaysLoad === true) return false

  // 2) non_deferrable builtins (GrowthBook + settings); default empty list
  if (getNonDeferrableBuiltins().includes(tool.name)) return false

  // 3) all MCP tools are deferred
  if (tool.isMcp === true) return true

  // 4) ToolSearch itself never deferred
  if (tool.name === "ToolSearch") return false

  // 5) Agent + fork-subagent enabled => never deferred
  if (tool.name === "Agent") {
    // (amt(), Qr(xYu)).isForkSubagentEnabled()
    if (isForkSubagentEnabled()) return false
    // NOTE: if fork disabled, fall through — Agent may still defer via shouldDefer
  }

  // 6) Brief / SendUserMessage (BRIEF_TOOL_NAME === "SendUserMessage") never deferred
  if (tool.name === "SendUserMessage") return false

  // 7) SendUserFile never deferred
  if (tool.name === "SendUserFile") return false

  // 8) PushNotification non-deferred only under remote_trigger entrypoint
  if (tool.name === "PushNotification" && isRemoteTriggerEntrypoint()) return false

  // 9) ScheduleWakeup non-deferred only when kairos loop dynamic GB is on
  if (tool.name === "ScheduleWakeup" && isKairosLoopDynamic()) return false

  if (tool.name === "EnterWorktree" && process.env.CLAUDE_CODE_SESSION_KIND === "bg") return false

  // 11) default: shouldDefer === true
  return tool.shouldDefer === true
}
