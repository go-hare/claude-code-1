  function onGestureChange(e){

  // Parity for the wheel/gesture* forwarding above, which covers only
  // mouse and trackpad input: on TOUCH devices this frame otherwise has
  // no pan/zoom path at all \u2014 the drag engine is mouse-event-only, so a
  // one-finger drag rubber-bands/scrolls the page and a two-finger
  // pinch drives the BROWSER's page zoom instead of the canvas. While
  // the parent canvas owns pan/zoom (hostPanZoom), the iframe root
  // carries inline touch-action:none so the browser never starts a
  // native scroll or pinch-zoom from a touch landing here (an inner
  // scrollable element keeps its own touch pan: the root is consulted
  // only for gestures it would handle itself), and the handlers below
  // forward one-finger pans and two-finger pinches through the same
  // rAF-coalesced panZoom channel the wheel path uses \u2014 the host
  // re-validates and per-event clamps every field either way. The style
  // RELAXES whenever native behavior should win instead: the interact
  // tool (live content owns its gestures \u2014 onWheel's bail) and inline
  // text editing (selection handles, magnifier, and the keyboard's
  // scroll-into-view must stay native).
  function touchForwardActive(){

  function syncTouchAction(){

  // Tracked touch contacts (id -> points). The first two drive the
  // gesture; later fingers just keep their coordinates fresh. A small
  // slop keeps taps clean: the compat mouse events that deliver tap
  // selection still fire untouched, and a pan only engages once the
  // finger has clearly moved.
  //
  // COORDINATES are the whole trick here. clientX/Y inside this frame
  // are the artboard's INTERNAL px, measured against an origin that
  // every forwarded pan/zoom MOVES \u2014 a motionless finger appears to
  // travel as the camera responds, which fed the forwarded deltas
  // back into themselves (pans settled at HALF the finger's motion,
  // pinches pinned at their equilibrium). screenX/Y is the stable
  // frame: it's glass-anchored, so neither the iframe's motion nor
  // its CSS scale moves a stationary finger's sample. All DELTAS and
  // the pinch spread are computed in screen space; only the ANCHOR
  // point forwards in internal px (clientX/Y), because the host maps
  // anchors through the artboard spec \u2014 the same contract the wheel
  // path uses. Screen-space deltas are glass px, which is the wheel
  // convention too, so they drop into the same channel unchanged.
  var touchPts = {}, touchIds = [], touchPanLive = false;
  var TOUCH_PAN_SLOP = 6;
  function onTouchDown(e){
    if (e.pointerType !== 'touch' || !touchForwardActive()) return;
    if (touchIds.indexOf(e.pointerId) === -1) touchIds.push(e.pointerId);
    // ix/iy: the contact's last INTERNAL-px position \u2014 kept alongside
    // the glass samples so the pinch can anchor at the PAIR's
    // midpoint (anchoring at whichever finger fired the event put the
    // focal point half the separation off-center, jittering between
    // contacts as their events interleave).
    touchPts[e.pointerId] = { x: e.screenX, y: e.screenY, sx: e.screenX, sy: e.screenY, ix: e.clientX, iy: e.clientY };
    // Membership changed \u2014 a pan in progress must not continue under a
    // joining finger (the pinch math re-bases from fresh coordinates).
    touchPanLive = false;
  }
  function onTouchMove(e){
    var p = touchPts[e.pointerId];
    if (!p || !touchForwardActive()) return;
    var gx = e.screenX, gy = e.screenY;
    p.ix = e.clientX; p.iy = e.clientY;
    var idx = touchIds.indexOf(e.pointerId);
    if (idx > 1) { p.x = gx; p.y = gy; return; }
    if (touchIds.length >= 2) {
      // Pinch: pan by the pair midpoint's screen motion, zoom by the
      // screen spread ratio in LOG space with the wheel path's
      // per-event clamp (the host clamps and budget-caps again).
      // Increments of a physical quantity telescope \u2014 a clamped tick
      // loses a little zoom rather than desyncing anything.
      var oid = touchIds[idx === 0 ? 1 : 0];
      var o = touchPts[oid];
      if (!o) return;
      var beforeMidX = (p.x + o.x) / 2, beforeMidY = (p.y + o.y) / 2;
      var beforeDist = Math.hypot(p.x - o.x, p.y - o.y);
      p.x = gx; p.y = gy;
      var afterMidX = (p.x + o.x) / 2, afterMidY = (p.y + o.y) / 2;
      var afterDist = Math.hypot(p.x - o.x, p.y - o.y);
      var dz = 0;
      if (beforeDist > 0 && afterDist > 0) {
        dz = Math.max(-0.22314, Math.min(0.22314, Math.log(afterDist / beforeDist)));
      }
      // Same sign convention as onWheel: the host pans by the NEGATED
      // forwarded delta, so content follows the fingers. The anchor is
      // the PAIR's midpoint in INTERNAL px (each contact's own latest
      // sample \u2014 see the header note): the point that should hold
      // still on screen while the spread changes around it.
      fwdPanZoom(true, beforeMidX - afterMidX, beforeMidY - afterMidY, dz, (p.ix + o.ix) / 2, (p.iy + o.iy) / 2, true);
      return;
    }
    // One finger: a canvas pan once past the slop.
    if (!touchPanLive) {
      if (Math.hypot(gx - p.sx, gy - p.sy) < TOUCH_PAN_SLOP) {
        p.x = gx; p.y = gy;
        return;
      }
      touchPanLive = true;
    }
    var dx = p.x - gx, dy = p.y - gy;
    p.x = gx; p.y = gy;
    if (dx !== 0 || dy !== 0) fwdPanZoom(false, dx, dy, 0, e.clientX, e.clientY, true);
  }
  function onTouchEnd(e){
    if (touchPts[e.pointerId] === undefined) return;
    delete touchPts[e.pointerId];
    var i = touchIds.indexOf(e.pointerId);
    if (i !== -1) touchIds.splice(i, 1);
    touchPanLive = false;
  }

  function touchKeepsNativeGesture(target){
    if (hostScroll && pageScrolls()) return true;
    var node = target;
    while (node && node !== document.documentElement) {
      if (node.nodeType === 1) {
        var tag = node.tagName;
        // contenteditable rides the text-entry carve-out (house
        // convention): selection-handle drags must stay native. Inline
        // artboard editing is additionally covered by the editing gate
        // in touchForwardActive.
        if (tag === 'INPUT' || tag === 'SELECT' || tag === 'TEXTAREA' || node.isContentEditable === true) return true;
        var cs = null;
        try { cs = window.getComputedStyle(node); } catch (err) { cs = null; }
        if (cs &&
            (((cs.overflowY === 'auto' || cs.overflowY === 'scroll') && node.scrollHeight > node.clientHeight) ||
             ((cs.overflowX === 'auto' || cs.overflowX === 'scroll') && node.scrollWidth > node.clientWidth))) {
          return true;
        }
      }
      node = node.parentElement;
    }
    return false;
  }
  function onSwallowTouchStart(e){
    if (!swallowArmed()) return;
    for (var i = 0; i < e.changedTouches.length; i++) {
      var t = e.changedTouches[i];
      swallowTouches[t.identifier] = !touchKeepsNativeGesture(t.target);
    }
  }
  function onSwallowTouchMove(e){
    if (!swallowArmed() || !e.cancelable) return;
    var n = e.changedTouches.length;
    if (n === 0) return;
    for (var i = 0; i < n; i++) {
      // Unknown identifiers (a touch predating the arm) default to NOT
      // swallowing \u2014 native behavior is the safe side.
      if (swallowTouches[e.changedTouches[i].identifier] !== true) return;
    }
    e.preventDefault();
  }
  function onSwallowTouchEnd(e){
    for (var i = 0; i < e.changedTouches.length; i++) {
      delete swallowTouches[e.changedTouches[i].identifier];
    }
  }

  function onDrop(e){
    if (!enabled) return;
    // Click-through mode: the page handles its own events.
    if (tool === 'interact') return;
    var dt = e.dataTransfer;
    if (!dt || !dt.files || dt.files.length === 0) return;
    e.preventDefault();
    e.stopPropagation();
    if (deckChromeAlongPath(e) && !railThumbAt(e.clientX, e.clientY)) return;
    var files = [];
    for (var i = 0; i < dt.files.length; i++) {
      if ((dt.files[i].type || '').indexOf('image/') === 0) files.push(dt.files[i]);
    }
    if (files.length === 0) return;
    var plan = fileDropPlan(e.clientX, e.clientY);
    if (plan && plan.refuse) return;
    var hovered = plan ? plan.el : null;
    post({
      type: 'fileDrop',
      files: files,
      name: hovered ? elOwner(hovered) : null,
      targetTid: hovered ? parseInt(hovered.getAttribute('data-dc-tpl'), 10) : null,
      position: plan ? plan.position : 'root-end',
      slide: hovered ? slideAnchorFor(hovered) : null,
      point: plan ? plan.point : null,
    });
    if (hovered) navigateToDropSlide(slideOf(hovered));
  }
