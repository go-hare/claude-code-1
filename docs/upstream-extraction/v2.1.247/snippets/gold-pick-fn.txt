function qhe(e){
  if(e.length===0)return;
  if(e.length===1)return e[0];
  let t=e.map((o)=>({tip:o,sessions:IV(o.id)}));
  return t.sort((o,n)=>{
    if(o.sessions!==n.sessions)return n.sessions-o.sessions;
    return(n.tip.priority??0)-(o.tip.priority??0)
  }),t[0]?.tip
}

async function Vhe(e){
  if(MV().spinnerTipsEnabled===!1)return;
  let t=await Whe(e);
  if(t.length===0)return;
  return qhe(t)
}
